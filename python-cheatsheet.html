🗂️Data Structures

Lists

Dictionaries

# Create dict person = { "name": "Alice", "age": 25, "city": "NYC" } # Access values print(person["name"]) # Alice print(person.get("age")) # 25 # Dict methods person["job"] = "Engineer" print(person.keys()) # dict_keys print(person.values()) # dict_values print(person.items()) # dict_items

🔤String Operations

String Methods

text = "Hello World" # Case operations print(text.upper()) # HELLO WORLD print(text.lower()) # hello world print(text.title()) # Hello World # Search & replace print(text.find("World")) # 6 print(text.replace("World", "Python")) # Split & join words = text.split() # ['Hello', 'World'] joined = " ".join(words) # "Hello World"

String Formatting

name = "Alice" age = 25 # f-strings (Python 3.6+) print(f"Hello {name}, you are {age} years old") # format() method print("Hello {}, you are {} years old".format(name, age)) # % formatting print("Hello %s, you are %d years old" % (name, age)) # String concatenation print(name + " is " + str(age) + " years old")

⚡List Comprehensions & Generators

List Comprehensions

# Basic comprehension squares = [x**2 for x in range(10)] # With condition even_squares = [x**2 for x in range(10) if x % 2 == 0] # Nested comprehension matrix = [[i*j for j in range(3)] for i in range(3)] # Dictionary comprehension squares_dict = {x: x**2 for x in range(5)} # Set comprehension squares_set = {x**2 for x in range(5)}

Generators

# Generator function def count_up_to(n): i = 0 while i < n: yield i i += 1 # Generator expression squares_gen = (x**2 for x in range(10)) # Using generators for num in count_up_to(5): print(num) # Convert to list nums_list = list(count_up_to(5))

Writing Files

# Write to file with open('output.txt', 'w') as f: f.write("Hello World\n") # Append to file with open('output.txt', 'a') as f: f.write("More content\n") # Write multiple lines lines = ["Line 1", "Line 2", "Line 3"] with open('output.txt', 'w') as f: f.writelines(line + '\n' for line in lines)

🧮Common Algorithms

Sorting

nums = [3, 1, 4, 1, 5] # Sort in place nums.sort() # [1, 1, 3, 4, 5] # Sort descending nums.sort(reverse=True) # Sort with key words = ["apple", "Banana", "cherry"] words.sort(key=str.lower) # Sorted (returns new list) sorted_nums = sorted(nums)

Searching

nums = [1, 2, 3, 4, 5] # Find index print(nums.index(3)) # 2 # Check existence print(3 in nums) # True # Binary search (manual) def binary_search(arr, target): left, right = 0, len(arr) - 1 while left <= right: mid = (left + right) // 2 if arr[mid] == target: return mid elif arr[mid] < target: left = mid + 1 else: right = mid - 1 return -1

🚨Exception Handling

Try-Except Blocks

try: # Code that might raise exception result = 10 / 0 except ZeroDivisionError: print("Cannot divide by zero") except ValueError: print("Invalid value") else: print("No exception occurred") finally: print("Always executed") # Catch all exceptions try: risky_code() except Exception as e: print(f"Error: {e}")

Custom Exceptions

class CustomError(Exception): def __init__(self, message): self.message = message super().__init__(self.message) def validate_age(age): if age < 0: raise CustomError("Age cannot be negative") if age > 150: raise ValueError("Age too high") try: validate_age(-5) except CustomError as e: print(e.message)
