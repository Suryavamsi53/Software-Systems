func maxSumSubarray(arr []int, k int) int {
  n := len(arr)
  if n < k {
    return 0
  }
  
  maxSum := 0
  windowSum := 0
  
  // Calculate sum of first window
  for i := 0; i < k; i++ {
    windowSum += arr[i]
  }
  maxSum = windowSum
  
  // Slide the window
  for i := k; i < n; i++ {
    windowSum = windowSum - arr[i-k] + arr[i]
    if windowSum > maxSum {
      maxSum = windowSum
    }
  }
  
  return maxSum
}
