func twoPointersSameDirection(arr []int) []int {
  n := len(arr)
  left := 0
  result := []int{}
  
  for right := 0; right < n; right++ {
    // Expand window with right pointer
    for left <= right && /* condition */ {
      left++ // Shrink window from left
    }
    
    // Process current window [left, right]
    result = append(result, arr[left:right+1]...)
  }
  
  return result
}
