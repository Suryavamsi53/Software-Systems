type Edge struct {
  u, v, weight int
}
func kruskal(edges []Edge, n int) []Edge {
  // Sort edges by weight
  sort.Slice(edges, func(i, j int) bool {
    return edges[i].weight < edges[j].weight
  })
  
  parent := make([]int, n)
  for i := range parent {
    parent[i] = i
  }
  
  mst := []Edge{}
  
  for _, edge := range edges {
    pu := find(parent, edge.u)
    pv := find(parent, edge.v)
    
    if pu != pv {
      union(parent, pu, pv)
      mst = append(mst, edge)
    }
  }
  
  return mst
}
